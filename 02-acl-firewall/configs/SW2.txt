! SW2 - Cisco 2960 - Zone-Based ACL Firewall (Project 2)
hostname SW2
no ip domain-lookup
ip domain-name office.lab
banner motd #Authorised access only. Activity is monitored.#
enable secret <REDACTED>
service password-encryption
security passwords min-length 10
username admin privilege 15 secret <REDACTED>
!
vlan 10
 name STAFF
vlan 20
 name ADMIN
vlan 30
 name GUEST
vlan 40
 name SERVERS
vlan 99
 name MGMT
vlan 666
 name PARKING
vlan 999
 name NATIVE
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
login block-for 120 attempts 3 within 60
!
ip dhcp snooping
ip dhcp snooping vlan 10,20,30
no ip dhcp snooping information option
ip arp inspection vlan 10,20,30
!
interface g0/1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,99,999
 switchport nonegotiate
 ip dhcp snooping trust
 ip arp inspection trust
 no shutdown
!
interface fa0/1
 switchport mode access
 switchport access vlan 10
interface range fa0/2 - 3
 switchport mode access
 switchport access vlan 30
interface range fa0/1 - 3
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
!
interface range fa0/4 - 24 , g0/2
 switchport mode access
 switchport access vlan 666
 shutdown
!
interface vlan 99
 ip address 192.168.99.12 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
!

 remark Only the Admin VLAN may manage devices
 permit 192.168.20.0 0.0.0.255
 deny any
!
no ip http server
no cdp run
!
line console 0
 login local

 logging synchronous

 access-class MGMT-ACCESS in
