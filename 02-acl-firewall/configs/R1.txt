! R1 - Cisco 2911 - Zone-Based ACL Firewall (Project 2)
hostname R1
no ip domain-lookup
ip domain-name office.lab
banner motd #Authorised access only. Activity is monitored.#
enable secret <REDACTED>
service password-encryption
security passwords min-length 10
username admin privilege 15 secret <REDACTED>
!
aaa new-model
aaa authentication login default local
aaa authorization exec default local
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
login block-for 120 attempts 3 within 60
!
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp pool STAFF
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.40.10
ip dhcp pool ADMIN
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.40.10
ip dhcp pool GUEST
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.40.10
!
interface g0/0
 no shutdown
interface g0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group STAFF-IN in
interface g0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
interface g0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip access-group GUEST-IN in
interface g0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip access-group SERVERS-IN in
interface g0/0.99
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
interface g0/0.999
 encapsulation dot1Q 999 native
interface g0/1
 description Link to ISP
 ip address 203.0.113.1 255.255.255.252
 no shutdown
!
ip route 0.0.0.0 0.0.0.0 203.0.113.2
!
ip access-list extended STAFF-IN
 remark Allow DHCP requests to R1
 permit udp any any eq bootps
 remark Allow ping to own gateway for troubleshooting
 permit icmp 192.168.10.0 0.0.0.255 host 192.168.10.1 echo
 remark Allow DNS, web and ping to the server
 permit udp 192.168.10.0 0.0.0.255 host 192.168.40.10 eq domain
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.40.10 eq www
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.40.10 eq 443
 permit icmp 192.168.10.0 0.0.0.255 host 192.168.40.10 echo
 remark Let Admin's pings to staff get a reply
 permit icmp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255 echo-reply
 remark Block all other internal destinations
 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
 remark Everything else (the Internet) is allowed
 permit ip 192.168.10.0 0.0.0.255 any
!
ip access-list extended GUEST-IN
 remark Allow DHCP requests to R1
 permit udp any any eq bootps
 remark Allow DNS only to the server
 permit udp 192.168.30.0 0.0.0.255 host 192.168.40.10 eq domain
 remark Block every internal network
 deny ip 192.168.30.0 0.0.0.255 192.168.0.0 0.0.255.255
 remark Internet only
 permit ip 192.168.30.0 0.0.0.255 any
!
ip access-list extended SERVERS-IN
 remark TCP replies only (ACK or RST bit set)
 permit tcp 192.168.40.0 0.0.0.255 any established
 remark DNS replies
 permit udp host 192.168.40.10 eq domain any
 remark Ping replies
 permit icmp 192.168.40.0 0.0.0.255 any echo-reply
 deny ip any any
!
ip access-list standard MGMT-ACCESS
 remark Only the Admin VLAN may manage devices
 permit 192.168.20.0 0.0.0.255
 deny any
!
no ip http server
no cdp run
!
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 login local
 transport input ssh
 access-class MGMT-ACCESS in
 exec-timeout 5 0
!
end
